cmake_minimum_required(VERSION 3.22)
project(PedalboardWindowsTestPlugins VERSION 1.0.0)

set(PEDALBOARD_JUCE_DIR "" CACHE PATH "Path to pedalboard's JUCE submodule")
set(PLUGINS_FILE "" CACHE FILEPATH "Plugin definitions, generated by build.ps1 from each plugin's .jucer file")

add_subdirectory("${PEDALBOARD_JUCE_DIR}" JUCE)

# Adds a VST3 plugin target. PLUGIN_OPTIONS are passed to juce_add_plugin, and
# MODULES are JUCE modules (juce_*) or paths to a plugin's own modules.
function(pedalboard_add_test_plugin target)
    cmake_parse_arguments(PARSE_ARGV 1 ARG "" ""
        "PLUGIN_OPTIONS;MODULES;SOURCES;RESOURCES;DEFINITIONS;COMPILE_OPTIONS")

    juce_add_plugin(${target} ${ARG_PLUGIN_OPTIONS} FORMATS VST3 VST3_AUTO_MANIFEST FALSE)
    target_sources(${target} PRIVATE ${ARG_SOURCES})

    if(ARG_RESOURCES)
        juce_add_binary_data(${target}Data SOURCES ${ARG_RESOURCES})
        target_link_libraries(${target} PRIVATE ${target}Data)
    endif()

    foreach(module IN LISTS ARG_MODULES)
        if(module MATCHES "^juce_")
            target_link_libraries(${target} PRIVATE juce::${module})
        else()
            juce_add_module("${module}")
            get_filename_component(module_name "${module}" NAME)
            target_link_libraries(${target} PRIVATE ${module_name})
        endif()
    endforeach()

    target_compile_definitions(${target} PUBLIC
        # Features that aren't needed by any of the test plugins:
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        ${ARG_DEFINITIONS})
    target_compile_options(${target} PRIVATE ${ARG_COMPILE_OPTIONS})
    target_link_libraries(${target} PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags)
    juce_generate_juce_header(${target})
endfunction()

include("${PLUGINS_FILE}")
